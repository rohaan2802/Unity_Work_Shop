using UnityEngine;

public class CharacterController : MonoBehaviour
{
    // Start is called once before the first execution of Update after the MonoBehaviour is created
    void Start()
    {
        
    }

    // Update is called once per frame
    void Update()
    {
        if (Input.GetAxis("Horizontal")>-2)
        {
            transform.position += new Vector3(Input.GetAxis("Horizontal") * Time.deltaTime, 0, 0);
        }

        if (Input.GetAxis("Vertical") > -2)
        {
            transform.position += new Vector3(0, 0, Input.GetAxis("Vertical") * Time.deltaTime);
        }
    }
}

Walker

Assets/Scripts/Courtyard/Walker.cs 

using UnityEngine;

/// <summary>
/// Walks with the legacy Input Manager. The workshop scripts named a class
/// CharacterController and moved by writing transform.position whenever the
/// axis was greater than -2. That test is true for every real axis value, and
/// writing the position ignores colliders. This component uses the engine
/// CharacterController and the axis value itself.
/// </summary>
[RequireComponent(typeof(CharacterController))]
public class Walker : MonoBehaviour
{
    public float speed = CourtyardLayout.MoveSpeed;
    public float gravity = CourtyardLayout.Gravity;
    public Vector3 spawn = CourtyardLayout.Spawn;

    CharacterController body;
    float verticalVelocity;
    bool returnPending;

    void Awake()
    {
        body = GetComponent<CharacterController>();
    }

    void Update()
    {
        if (returnPending)
        {
            returnPending = false;
            ApplySpawn();
        }

        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");
        Vector3 wish = new Vector3(horizontal, 0f, vertical);
        if (wish.sqrMagnitude > 1f)
        {
            wish.Normalize();
        }

        if (body.isGrounded && verticalVelocity < 0f)
        {
            verticalVelocity = -1f;
        }

        verticalVelocity += gravity * Time.deltaTime;

        Vector3 motion = wish * speed;
        motion.y = verticalVelocity;
        body.Move(motion * Time.deltaTime);

        // Trigger messages arrive inside Move. The pad only sets a flag.
        // Teleporting here is after Move has returned.
        if (returnPending || transform.position.y < -4f)
        {
            returnPending = false;
            ApplySpawn();
            return;
        }

        if (wish.sqrMagnitude > 0.0001f)
        {
            Quaternion face = Quaternion.LookRotation(wish, Vector3.up);
            float turn = 1f - Mathf.Exp(-10f * Time.deltaTime);
            transform.rotation = Quaternion.Slerp(transform.rotation, face, turn);
        }
    }

    /// <summary>
    /// Used by the reset pad. Safe to call from OnTriggerEnter because the
    /// actual teleport waits until Move has returned.
    /// </summary>
    public void RequestSpawnReturn()
    {
        returnPending = true;
    }

    /// <summary>
    /// Used by the R-key restart, which runs from Update and is not inside Move.
    /// </summary>
    public void ApplySpawn()
    {
        if (body == null)
        {
            body = GetComponent<CharacterController>();
        }

        body.enabled = false;
        transform.SetPositionAndRotation(spawn, Quaternion.identity);
        body.enabled = true;
        verticalVelocity = 0f;
        returnPending = false;
    }
}

using UnityEngine;

/// <summary>
/// Shared sizes for Courtyard Walk. The browser concept illustration copies these
/// numbers so the ground-plane rules match. It is not a Unity build.
/// </summary>
public static class CourtyardLayout
{
    public const float ArenaHalf = 12f;
    public const float WallThickness = 0.5f;
    public const float PlayerRadius = 0.5f;
    public const float MoveSpeed = 5f;
    public const float Gravity = -20f;
    public const float OrbRadius = 0.45f;
    public const float ColumnRadius = 0.6f;

    public static readonly Vector3 Spawn = new Vector3(0f, 1f, -8f);
    public static readonly Vector3 PadCenter = new Vector3(8f, 0.75f, -8f);
    public static readonly Vector3 PadScale = new Vector3(2.4f, 1.5f, 2.4f);
    public static readonly Vector3 ColumnCenter = new Vector3(3f, 1f, -4f);

    public static readonly Vector3[] Orbs =
    {
        new Vector3(-8f, 0.7f, -6f),
        new Vector3(8f, 0.7f, -4f),
        new Vector3(7f, 0.7f, 7f),
        new Vector3(-9f, 0.7f, 1f),
        new Vector3(1.5f, 0.7f, 0.5f)
    };

    public static readonly Vector3[] Blocks =
    {
        new Vector3(4f, 1f, 2f),
        new Vector3(-5f, 1f, 5f),
        new Vector3(-1f, 1f, 8f),
        new Vector3(-3.2f, 0.5f, -2f)
    };

    public static readonly Vector3[] BlockScales =
    {
        new Vector3(2f, 2f, 2f),
        new Vector3(3f, 2f, 1.6f),
        new Vector3(2f, 2f, 2f),
        new Vector3(2.2f, 1f, 2.2f)
    };
}

CourtyardBuild

Assets/Scripts/Courtyard/CourtyardBuild.cs 

using System.Collections.Generic;
using UnityEngine;

/// <summary>
/// Builds the courtyard from primitives. Play mode and the editor menu both call Create.
/// </summary>
public static class CourtyardBuild
{
    public static GameObject Create()
    {
        GameObject root = new GameObject("CourtyardRoot");
        CourtyardSession session = root.AddComponent<CourtyardSession>();
        CourtyardHud hud = root.AddComponent<CourtyardHud>();
        hud.session = session;

        CreateGround(root.transform);
        CreateWalls(root.transform);
        CreateBlocks(root.transform);
        CreateColumn(root.transform);
        List<GameObject> orbs = CreateOrbs(root.transform, session);
        CreatePad(root.transform, session);
        Walker walker = CreateWalker(root.transform);
        session.Configure(walker, orbs);
        AttachCamera(walker.transform);
        TuneLight();
        return root;
    }

    static void CreateGround(Transform parent)
    {
        GameObject ground = GameObject.CreatePrimitive(PrimitiveType.Cube);
        ground.name = "Ground";
        ground.transform.SetParent(parent, false);
        float span = CourtyardLayout.ArenaHalf * 2f + 1f;
        ground.transform.position = new Vector3(0f, -0.5f, 0f);
        ground.transform.localScale = new Vector3(span, 1f, span);
        Paint(ground, new Color(0.76f, 0.66f, 0.48f));
    }

    static void CreateWalls(Transform parent)
    {
        float edge = CourtyardLayout.ArenaHalf;
        float span = CourtyardLayout.ArenaHalf * 2f;
        float thick = CourtyardLayout.WallThickness;
        float height = 2.2f;
        float y = height * 0.5f;
        Wall(parent, "Wall North", new Vector3(0f, y, edge), new Vector3(span, height, thick));
        Wall(parent, "Wall South", new Vector3(0f, y, -edge), new Vector3(span, height, thick));
        Wall(parent, "Wall East", new Vector3(edge, y, 0f), new Vector3(thick, height, span));
        Wall(parent, "Wall West", new Vector3(-edge, y, 0f), new Vector3(thick, height, span));
    }

    static void Wall(Transform parent, string name, Vector3 position, Vector3 scale)
    {
        GameObject wall = GameObject.CreatePrimitive(PrimitiveType.Cube);
        wall.name = name;
        wall.transform.SetParent(parent, false);
        wall.transform.position = position;
        wall.transform.localScale = scale;
        Paint(wall, new Color(0.45f, 0.38f, 0.32f));
    }

    static void CreateBlocks(Transform parent)
    {
        for (int i = 0; i < CourtyardLayout.Blocks.Length; i++)
        {
            GameObject block = GameObject.CreatePrimitive(PrimitiveType.Cube);
            block.name = "Block " + (i + 1);
            block.transform.SetParent(parent, false);
            block.transform.position = CourtyardLayout.Blocks[i];
            block.transform.localScale = CourtyardLayout.BlockScales[i];
            Paint(block, new Color(0.55f, 0.47f, 0.39f));
        }
    }

    static void CreateColumn(Transform parent)
    {
        GameObject column = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
        column.name = "Column";
        column.transform.SetParent(parent, false);
        column.transform.position = CourtyardLayout.ColumnCenter;
        column.transform.localScale = new Vector3(CourtyardLayout.ColumnRadius * 2f, 1f, CourtyardLayout.ColumnRadius * 2f);
        Paint(column, new Color(0.63f, 0.56f, 0.47f));
    }

    static List<GameObject> CreateOrbs(Transform parent, CourtyardSession session)
    {
        List<GameObject> list = new List<GameObject>();
        for (int i = 0; i < CourtyardLayout.Orbs.Length; i++)
        {
            GameObject orb = GameObject.CreatePrimitive(PrimitiveType.Sphere);
            orb.name = "Orb " + (i + 1);
            orb.transform.SetParent(parent, false);
            orb.transform.position = CourtyardLayout.Orbs[i];
            orb.transform.localScale = Vector3.one * (CourtyardLayout.OrbRadius * 2f);
            SphereCollider collider = orb.GetComponent<SphereCollider>();
            collider.isTrigger = true;
            Rigidbody body = orb.AddComponent<Rigidbody>();
            body.isKinematic = true;
            body.useGravity = false;
            orb.AddComponent<CollectibleOrb>().Bind(session);
            Paint(orb, new Color(0.93f, 0.76f, 0.28f));
            list.Add(orb);
        }

        return list;
    }

    static void CreatePad(Transform parent, CourtyardSession session)
    {
        GameObject pad = GameObject.CreatePrimitive(PrimitiveType.Cube);
        pad.name = "Reset Pad";
        pad.transform.SetParent(parent, false);
        pad.transform.position = CourtyardLayout.PadCenter;
        pad.transform.localScale = CourtyardLayout.PadScale;
        BoxCollider collider = pad.GetComponent<BoxCollider>();
        collider.isTrigger = true;
        Rigidbody body = pad.AddComponent<Rigidbody>();
        body.isKinematic = true;
        body.useGravity = false;
        pad.AddComponent<ResetPad>().Bind(session);
        Paint(pad, new Color(0.72f, 0.22f, 0.18f));
    }

    static Walker CreateWalker(Transform parent)
    {
        GameObject player = GameObject.CreatePrimitive(PrimitiveType.Capsule);
        player.name = "Walker";
        player.transform.SetParent(parent, false);
        player.transform.position = CourtyardLayout.Spawn;
        // The primitive brings its own capsule collider. Destroy it immediately.
        // Waiting until end of frame would leave two colliders on the walker.
        CapsuleCollider primitiveCollider = player.GetComponent<CapsuleCollider>();
        if (primitiveCollider != null)
        {
            Object.DestroyImmediate(primitiveCollider);
        }
        CharacterController controller = player.AddComponent<CharacterController>();
        controller.height = 2f;
        controller.radius = CourtyardLayout.PlayerRadius;
        controller.center = Vector3.zero;
        controller.stepOffset = 0.3f;
        Paint(player, new Color(0.1f, 0.42f, 0.44f));
        return player.AddComponent<Walker>();
    }

    static void AttachCamera(Transform target)
    {
        Camera camera = Camera.main;
        if (camera == null)
        {
            GameObject cameraObject = new GameObject("Main Camera");
            cameraObject.tag = "MainCamera";
            camera = cameraObject.AddComponent<Camera>();
            cameraObject.AddComponent<AudioListener>();
        }

        camera.clearFlags = CameraClearFlags.SolidColor;
        camera.backgroundColor = new Color(0.62f, 0.74f, 0.82f);
        FollowCamera follow = camera.GetComponent<FollowCamera>();
        if (follow == null)
        {
            follow = camera.gameObject.AddComponent<FollowCamera>();
        }

        follow.target = target;
        camera.transform.position = target.position + follow.offset;
        camera.transform.LookAt(target.position + Vector3.up * 1.2f);
    }

    static void TuneLight()
    {
        Light sun = Object.FindAnyObjectByType<Light>();
        if (sun == null)
        {
            GameObject lightObject = new GameObject("Directional Light");
            sun = lightObject.AddComponent<Light>();
            sun.type = LightType.Directional;
            lightObject.transform.rotation = Quaternion.Euler(50f, -30f, 0f);
        }

        sun.color = new Color(1f, 0.95f, 0.84f);
        sun.intensity = 1.15f;
        sun.shadows = LightShadows.Soft;
    }

    static void Paint(GameObject target, Color color)
    {
        Renderer renderer = target.GetComponent<Renderer>();
        if (renderer == null)
        {
            return;
        }

        Material material = renderer.material;
        if (material.HasProperty("_Color"))
        {
            material.color = color;
        }

        if (material.HasProperty("_BaseColor"))
        {
            material.SetColor("_BaseColor", color);
        }
    }

}

CourtyardSession

Assets/Scripts/Courtyard/CourtyardSession.cs 

using System.Collections.Generic;
using UnityEngine;

/// <summary>
/// Score and restart state for one walk of the courtyard.
/// </summary>
public class CourtyardSession : MonoBehaviour
{
    public int collected;
    public int total;
    public string status = "Walk the courtyard and touch every orb.";
    public bool cleared;

    readonly List<GameObject> orbs = new List<GameObject>();
    Walker walker;

    public void Configure(Walker player, List<GameObject> orbObjects)
    {
        walker = player;
        orbs.Clear();
        orbs.AddRange(orbObjects);
        total = orbs.Count;
        collected = 0;
        cleared = false;
        status = "Walk the courtyard and touch every orb.";
    }

    void Update()
    {
        if (Input.GetKeyDown(KeyCode.R))
        {
            Restart();
        }
    }

    public void RegisterCollection()
    {
        collected = Mathf.Clamp(collected + 1, 0, total);
        if (total > 0 && collected >= total)
        {
            cleared = true;
            status = "Courtyard clear. Press R to walk it again.";
        }
        else
        {
            status = "Orb collected.";
        }
    }

    public void NotifyPad(Walker who)
    {
        who.RequestSpawnReturn();
        if (cleared)
        {
            status = "Courtyard clear. The pad only returns you to the start.";
        }
        else
        {
            status = "Reset pad. Orbs you already took stay taken.";
        }
    }

    public void Restart()
    {
        collected = 0;
        cleared = false;
        status = "Restarted. The orbs are back.";
        for (int i = 0; i < orbs.Count; i++)
        {
            if (orbs[i] != null)
            {
                orbs[i].SetActive(true);
            }
        }

        if (walker != null)
        {
            walker.ApplySpawn();
        }
    }
}

using UnityEngine;

/// <summary>
/// Immediate-mode HUD. It uses the built-in IMGUI module so the scene does not
/// need a Canvas, a font asset, or TextMeshPro.
/// </summary>
public class CourtyardHud : MonoBehaviour
{
    public CourtyardSession session;

    void OnGUI()
    {
        if (session == null)
        {
            return;
        }

        float scale = Mathf.Max(1f, Screen.height / 900f);
        Matrix4x4 previous = GUI.matrix;
        GUI.matrix = Matrix4x4.TRS(Vector3.zero, Quaternion.identity, new Vector3(scale, scale, 1f));

        float width = Mathf.Min(540f, (Screen.width / scale) - 32f);
        GUIStyle title = new GUIStyle(GUI.skin.label);
        title.fontSize = 28;
        title.fontStyle = FontStyle.Bold;
        title.normal.textColor = Color.white;
        title.wordWrap = true;

        GUIStyle body = new GUIStyle(GUI.skin.label);
        body.fontSize = 20;
        body.wordWrap = true;
        body.normal.textColor = new Color(0.97f, 0.94f, 0.88f);

        Color previousColor = GUI.color;
        GUI.color = new Color(0.11f, 0.09f, 0.08f, 0.86f);
        GUI.Box(new Rect(16f, 16f, width, 230f), GUIContent.none);
        GUI.color = previousColor;

        float x = 32f;
        float y = 28f;
        GUI.Label(new Rect(x, y, width - 32f, 40f), "Courtyard Walk", title);
        y += 44f;
        GUI.Label(new Rect(x, y, width - 32f, 32f), "Orbs " + session.collected + " / " + session.total, body);
        y += 34f;
        GUI.Label(new Rect(x, y, width - 32f, 72f), session.status, body);
        y += 74f;
        GUI.Label(new Rect(x, y, width - 32f, 64f), "WASD or arrow keys. The red pad returns you to the start. R restarts.", body);

        GUI.matrix = previous;
    }
}

FollowCamera

Assets/Scripts/Courtyard/FollowCamera.cs 

using UnityEngine;

/// <summary>
/// Runs in LateUpdate so the camera follows the walker after Update has moved it.
/// LateUpdate still runs every frame. It is not a one-shot timer.
/// </summary>
public class FollowCamera : MonoBehaviour
{
    public Transform target;
    public Vector3 offset = new Vector3(0f, 7.5f, -9f);
    public float followSharpness = 6f;

    void LateUpdate()
    {
        if (target == null)
        {
            return;
        }

        Vector3 desired = target.position + offset;
        float blend = 1f - Mathf.Exp(-followSharpness * Time.deltaTime);
        transform.position = Vector3.Lerp(transform.position, desired, blend);
        transform.LookAt(target.position + Vector3.up * 1.2f);
    }
}

CollectibleOrb

Assets/Scripts/Courtyard/CollectibleOrb.cs 

using UnityEngine;

/// <summary>
/// Trigger orb. A kinematic Rigidbody is required so Unity sends OnTriggerEnter
/// when a CharacterController moves into the trigger.
/// </summary>
public class CollectibleOrb : MonoBehaviour
{
    CourtyardSession session;

    public void Bind(CourtyardSession owner)
    {
        session = owner;
    }

    void OnTriggerEnter(Collider other)
    {
        if (!isActiveAndEnabled)
        {
            return;
        }

        Walker walker = other.GetComponent<Walker>();
        if (walker == null)
        {
            walker = other.GetComponentInParent<Walker>();
        }

        if (walker == null)
        {
            return;
        }

        gameObject.SetActive(false);
        if (session != null)
        {
            session.RegisterCollection();
        }
    }
}

ResetPad

Assets/Scripts/Courtyard/ResetPad.cs 

using UnityEngine;

/// <summary>
/// Returns the walker to the spawn point. Collected orbs stay collected.
/// The R key, handled by CourtyardSession, is the full restart.
/// </summary>
public class ResetPad : MonoBehaviour
{
    CourtyardSession session;

    public void Bind(CourtyardSession owner)
    {
        session = owner;
    }

    void OnTriggerEnter(Collider other)
    {
        Walker walker = other.GetComponent<Walker>();
        if (walker == null)
        {
            walker = other.GetComponentInParent<Walker>();
        }

        if (walker == null || session == null)
        {
            return;
        }

        session.NotifyPad(walker);
    }
}

Editor scene builder

Assets/Editor/CourtyardSceneBuilder.cs 

using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;

/// <summary>
/// Bakes the same primitives the play-mode bootstrap creates, then saves the scene.
/// Play already works without this menu. Use it when you want the hierarchy visible
/// before entering Play mode.
/// </summary>
public static class CourtyardSceneBuilder
{
    const string ScenePath = "Assets/Scenes/CourtyardWalk.unity";

    [MenuItem("Workshop/Build Courtyard Walk Scene")]
    public static void BuildAndSave()
    {
        EditorSceneManager.OpenScene(ScenePath, OpenSceneMode.Single);
        GameObject existing = GameObject.Find("CourtyardRoot");
        if (existing != null)
        {
            Object.DestroyImmediate(existing);
        }

        CourtyardBuild.Create();
        EditorSceneManager.MarkSceneDirty(EditorSceneManager.GetActiveScene());
        EditorSceneManager.SaveScene(EditorSceneManager.GetActiveScene(), ScenePath);
        Debug.Log("Saved Courtyard Walk to " + ScenePath);
    }
}
